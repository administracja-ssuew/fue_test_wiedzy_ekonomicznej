**useWakeLock.js** (przepisz; komentarz po polsku: dlaczego gest, odesłanie do lib/wakeLock.js):
```js
import { useEffect, useSyncExternalStore } from "react";
import { getWakeLock } from "../lib/wakeLock.js";
const IDLE = { wanted: false, held: false, failed: false, mode: null };
const noopSubscribe = () => () => {};
export default function useWakeLock(active) {
  useEffect(() => { getWakeLock()?.setWanted(!!active); }, [active]);
}
export function useWakeLockState() {
  const wl = getWakeLock();
  return useSyncExternalStore(wl ? wl.subscribe : noopSubscribe, wl ? wl.getState : () => IDLE, () => IDLE);
}
```
